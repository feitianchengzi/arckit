package handler

import (
	"fmt"
	"github.com/gin-gonic/gin"
	"net/http/httptest"
	"strings"
	"sync"
	"testing"
	"todo/models"
)

func TestProjectMemberAddAuthorizationAndTransactions(t *testing.T) {
	db := openFeedbackWorkflowPostgres(t)
	if err := db.AutoMigrate(&models.Organization{}, &models.OrganizationMember{}, &models.ProjectEvent{}); err != nil {
		t.Fatal(err)
	}
	org := uint(31)
	project := models.Project{Name: "direct add", OrganizationID: &org, CreatorID: 7}
	if err := db.Create(&project).Error; err != nil {
		t.Fatal(err)
	}
	for i, role := range []string{"owner", "admin", "member"} {
		if err := db.Create(&models.ProjectMember{ProjectID: project.ID, UserID: uint(i + 7), Role: role}).Error; err != nil {
			t.Fatal(err)
		}
	}
	for _, m := range []models.OrganizationMember{{ID: 91, OrganizationID: 31, UserID: 17, Role: "member"}, {ID: 92, OrganizationID: 32, UserID: 18, Role: "member"}, {ID: 93, OrganizationID: 31, UserID: 10, Role: "owner"}, {ID: 94, OrganizationID: 31, UserID: 19, Role: "member"}} {
		if err := db.Create(&m).Error; err != nil {
			t.Fatal(err)
		}
	}
	request := func(actor uint, pid uint, member uint) int {
		w := httptest.NewRecorder()
		c, _ := gin.CreateTestContext(w)
		c.Set("db", db)
		if actor != 0 {
			c.Set("userID", actor)
		}
		c.Params = gin.Params{{Key: "id", Value: fmt.Sprint(pid)}}
		c.Request = httptest.NewRequest("POST", "/projects/members", strings.NewReader(fmt.Sprintf(`{"organization_member_id":%d,"role":"owner"}`, member)))
		c.Request.Header.Set("Content-Type", "application/json")
		AddProjectMember(c)
		return w.Code
	}
	for _, actor := range []uint{0, 9, 10, 11} {
		want := 403
		if actor == 0 {
			want = 401
		}
		if got := request(actor, project.ID, 91); got != want {
			t.Fatalf("actor %d: %d want %d", actor, got, want)
		}
	}
	if got := request(7, project.ID, 92); got != 400 {
		t.Fatalf("cross organization: %d", got)
	}
	if got := request(7, project.ID, 17); got != 404 {
		t.Fatalf("user id is not organization member id: %d", got)
	}
	personal := models.Project{Name: "personal", CreatorID: 7}
	if err := db.Create(&personal).Error; err != nil {
		t.Fatal(err)
	}
	if got := request(7, personal.ID, 91); got != 400 {
		t.Fatalf("personal: %d", got)
	}
	if got := request(7, project.ID, 91); got != 201 {
		t.Fatalf("owner create: %d", got)
	}
	var member models.ProjectMember
	if err := db.Where("project_id = ? AND user_id = ?", project.ID, 17).First(&member).Error; err != nil {
		t.Fatal(err)
	}
	if member.Role != "member" {
		t.Fatalf("role escalated: %s", member.Role)
	}
	db.Model(&member).Updates(map[string]any{"role": "admin", "duty": "retained", "is_external": true})
	if got := request(8, project.ID, 91); got != 200 {
		t.Fatalf("admin idempotent: %d", got)
	}
	db.First(&member, member.ID)
	if member.Role != "admin" || (member.Duty == nil || *member.Duty != "retained") || !member.IsExternal {
		t.Fatal("idempotent write changed relationship")
	}
	for _, actor := range []uint{9, 10, 11} {
		if got := request(actor, project.ID, 91); got != 403 {
			t.Fatalf("duplicate actor %d: %d", actor, got)
		}
	}
	var count int64
	db.Model(&models.ProjectEvent{}).Where("project_id = ?", project.ID).Count(&count)
	if count != 1 {
		t.Fatalf("duplicate events: %d", count)
	}
	var wg sync.WaitGroup
	codes := make(chan int, 8)
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func() { defer wg.Done(); codes <- request(8, project.ID, 94) }()
	}
	wg.Wait()
	close(codes)
	creates := 0
	for code := range codes {
		if code == 201 {
			creates++
		} else if code != 200 {
			t.Fatalf("concurrent add: %d", code)
		}
	}
	if creates != 1 {
		t.Fatalf("creates %d", creates)
	}
	db.Model(&models.ProjectEvent{}).Where("project_id = ?", project.ID).Count(&count)
	if count != 2 {
		t.Fatalf("events %d", count)
	}
	if err := db.Exec("DROP TABLE project_events").Error; err != nil {
		t.Fatal(err)
	}
	if got := request(7, project.ID, 93); got != 500 {
		t.Fatalf("event failure: %d", got)
	}
	db.Model(&models.ProjectMember{}).Where("project_id = ? AND user_id = ?", project.ID, 10).Count(&count)
	if count != 0 {
		t.Fatal("member was not rolled back with event")
	}
}
